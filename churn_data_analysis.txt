import pandas as pd
import numpy as np

df=pd.read_csv("/content/Churn_Modelling - Churn_Modelling.csv")
df.head()

#1. Find the total number of customers in each country.
total_customers=df.groupby("Geography")['CustomerId'].size().reset_index(name='count')
total_customers

#2. Find the average credit score of customers based on gender.
avg_creadit_score=df.groupby('Gender')['CreditScore'].mean()
avg_creadit_score

#3. Find the total number of customers who exited and customers who stayed.
df['Exited'].value_counts().reset_index(name='count')

#4. Find the average account balance in each country.
avg_account_balance=df.groupby('Geography')['Balance'].mean()
avg_account_balance

#5. Find which age has the highest number of customers who exited the bank.
age_wise_customers=df.groupby('Age')['Exited'].value_counts().reset_index(name='no.of_customers')
age_wise_customers

#6. Find the average estimated salary of active and inactive members.
avg_estimated_salary=df.groupby('IsActiveMember')['EstimatedSalary'].mean().reset_index(name='avg_salary')
avg_estimated_salary

#7. Find the churn count based on the number of products customers are using.
churn_count=df.groupby('NumOfProducts')['Exited'].value_counts().reset_index(name='churn_count')
churn_count

#8. Create a separate DataFrame containing country and region information, merge it with the
#main dataset using pd.merge(), and then find the average balance and average estimated
#salary by region.

import pandas as pd

# Define the regional mapping
region_data = {
    'Geography': ['France', 'Germany', 'Spain'],
    'Region': ['Western Europe', 'Central Europe', 'Southern Europe']
}

df_regions = pd.DataFrame(region_data)

# Merge the dataframes on the common 'Geography' column
df_merged = pd.merge(df, df_regions, on='Geography')
df_merged.head()

# Group by Region and calculate the mean for specific columns
regional_stats = df_merged.groupby('Region')[['Balance', 'EstimatedSalary']].mean()

print(regional_stats)
