import numpy as np
import pandas as pd

df=pd.read_csv("/content/Attrition - Attrition.csv")
df.head(2)

#!pip install plotly

import plotly.graph_objects as go
import plotly.io as pio
pio.templates.default="plotly_white"

print(df.isnull().sum())

attr_df=df[df['Attrition']=='Yes']
attr_df.head()

attr_by_dept=attr_df.groupby(['Department']).size().reset_index(name='count')
attr_by_dept.head()

fig = go.Figure(data=[go.Pie(
    labels=attr_by_dept['Department'],
    values=attr_by_dept['count'],
    hole=0.4,
    marker=dict(colors=['red','green']),
    textposition='inside'
)])

fig.update_layout(title='Attrition by Department')

fig.show()

df.columns

attr_by_educationfield=attr_df.groupby(['EducationField']).size().reset_index(name='count')
attr_by_educationfield

fig =go.Figure(data=[go.Pie(
    labels=attr_by_educationfield['EducationField'],
    values=attr_by_educationfield['count'],
    hole=0.4,
    marker=dict(colors=['red','green']),
    textposition='inside'
)])

fig.update_layout(title='Attrition by EducationField')

fig.show()

#we can see that the previous employees with life science as an education field have a  high attrition rate. now lets have
#a look at the percentage of attrition by number of years at the company.

attrition_by_noofyears=attr_df.groupby(['YearsAtCompany']).size().reset_index(name='count')
attrition_by_noofyears

#we can see that most of the employees leave the organization after completing a year

attr_by_promotion=attr_df.groupby(['YearsSinceLastPromotion']).size().reset_index(name='Count')
attr_by_promotion

#now lets have a look at the percentage of attrition by gender

attr_by_gender=attr_df.groupby(['Gender']).size().reset_index(name='Count')
attr_by_gender

#create a donut chart
fig =go.Figure(data=[go.Pie(
    labels=attr_by_gender['Gender'],

    textposition='inside'

)])

#update by layout
fig.update_layout(title='Attrition by gender')
